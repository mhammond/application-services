12impl Store {
13    /// Returns a bridged sync engine for addresses, for use by Desktop's Sync
14    /// framework. Constructing a `ConfigSyncEngine` only assembles structs and
15    /// never touches the DB, so this cannot fail.
16    pub fn addresses_bridged_engine(self: Arc<Self>) -> Arc<AddressesBridgedEngine> {
17        let engine = crate::sync::address::create_engine(self);
18        Arc::new(AddressesBridgedEngine::new(Box::new(
19            AddressesBridgedEngineAdaptor { engine },
20        )))
21    }
22}

24/// `ConfigSyncEngine` implements `sync15::SyncEngine`, which is what the sync
25/// manager drives. Desktop instead speaks `mozIBridgedSyncEngine`, whose Rust
26/// shape is `sync15::BridgedEngine`. The two differ only in that the bridge owns
27/// the last-sync timestamp explicitly, so this adaptor supplies that and the
28/// blanket `impl<A: BridgedEngineAdaptor> BridgedEngine for A` provides the rest.
29struct AddressesBridgedEngineAdaptor {
30    engine: ConfigSyncEngine<InternalAddress>,
31}
32
33impl BridgedEngineAdaptor for AddressesBridgedEngineAdaptor {
34    fn last_sync(&self) -> Result<i64> {
35        Ok(self.engine.get_last_sync_millis()?)
36    }
37
38    fn set_last_sync(&self, last_sync_millis: i64) -> Result<()> {
39        self.engine.set_last_sync_millis(last_sync_millis)?;
40        Ok(())
41    }
42
43    fn engine(&self) -> &dyn sync15::engine::SyncEngine {
44        &self.engine
45    }
46}

51sync15::uniffi_bridged_engine!(AddressesBridgedEngine, String);

53#[cfg(test)]
54mod tests {
55    use super::*;
56    use crate::db::models::address::UpdatableAddressFields;
57    use std::collections::HashMap;

59    // Exercises the sync metadata the bridge owns: last_sync, sync_id and reset.
60    #[test]
61    fn test_sync_meta() {
62        error_support::init_for_tests();
63
64        let store = Arc::new(Store::new_shared_memory("addresses-bridge").unwrap());
65        let bridge = store.addresses_bridged_engine();
66
67        // Fresh DB: never synced.
68        assert_eq!(bridge.last_sync().unwrap(), 0);
69        bridge.set_last_sync(3).unwrap();
70        assert_eq!(bridge.last_sync().unwrap(), 3);
71
72        assert!(bridge.sync_id().unwrap().is_none());
73
74        bridge.ensure_current_sync_id("some_guid").unwrap();
75        assert_eq!(bridge.sync_id().unwrap(), Some("some_guid".to_string()));
76        // changing the sync ID resets the timestamp
77        assert_eq!(bridge.last_sync().unwrap(), 0);
78        bridge.set_last_sync(3).unwrap();
79
80        bridge.reset_sync_id().unwrap();
81        assert_ne!(bridge.sync_id().unwrap(), Some("some_guid".to_string()));
82        assert_eq!(bridge.last_sync().unwrap(), 0);
83        bridge.set_last_sync(3).unwrap();
84
85        // `reset` clears the guid and the timestamp.
86        bridge.reset().unwrap();
87        assert_eq!(bridge.last_sync().unwrap(), 0);
88        assert!(bridge.sync_id().unwrap().is_none());
89    }

91    // A roundtrip through the bridge's data path: stage an incoming remote
92    // address, apply it, and confirm the local-only address comes back out for
93    // upload. Unlike `test_sync_meta` this exercises the JSON (de)serialization
94    // of BSOs and the sync staging tables, mirroring the logins and tabs
95    // `test_sync_via_bridge` tests.
96    #[test]
97    fn test_sync_via_bridge() {
98        error_support::init_for_tests();
99
100        let store = Arc::new(Store::new_shared_memory("addresses-bridge-roundtrip").unwrap());
101
102        // A local-only address: nothing on the server knows about it yet, so it
103        // should be uploaded.
104        let local = store
105            .add_address(UpdatableAddressFields {
106                name: "Local Person".to_string(),
107                street_address: "1 Local Lane".to_string(),
108                address_level2: "Seattle, WA".to_string(),
109                country: "US".to_string(),
110                ..Default::default()
111            })
112            .expect("should add local address");
113
114        let bridge = store.clone().addresses_bridged_engine();
115
116        // `prepare_for_sync` is what creates the sync staging tables; the client
117        // data it is given is unused by this engine.
118        bridge
119            .prepare_for_sync(r#"{"local_client_id":"my-client","recent_clients":{}}"#)
120            .expect("should prepare for sync");
121        bridge.sync_started().unwrap();
122
123        // An incoming remote address that isn't known locally. We build the
124        // envelope as raw JSON, exactly as the JS bridge hands it to us.
125        let incoming = vec![serde_json::json!({
126            "id": "remote-only-bbbb",
127            "modified": 0,
128            "payload": serde_json::json!({
129                "id": "remote-only-bbbb",
130                "entry": {
131                    "name": "Remote Person",
132                    "street-address": "99 Remote Road",
133                    "address-level2": "Portland, OR",
134                    "country": "US",
135                    "version": 1,
136                },
137            })
138            .to_string(),
139        })
140        .to_string()];
141        bridge
142            .store_incoming(incoming)
143            .expect("should store incoming");
144
145        // Applying stores the remote record locally and returns the local-only
146        // address for upload.
147        let outgoing = bridge.apply().expect("should apply");
148        let changes: HashMap<String, serde_json::Value> = outgoing
149            .into_iter()
150            .map(|s| {
151                let bso: serde_json::Value = serde_json::from_str(&s).unwrap();
152                let payload: serde_json::Value =
153                    serde_json::from_str(bso["payload"].as_str().unwrap()).unwrap();
154                (payload["id"].as_str().unwrap().to_string(), payload)
155            })
156            .collect();
157
158        // Only the local address is outgoing; the just-applied remote one is not
159        // re-uploaded.
160        assert_eq!(changes.len(), 1);
161        assert_eq!(
162            changes[&local.guid]["entry"]["street-address"],
163            "1 Local Lane"
164        );
165
166        // The incoming remote address was actually persisted.
167        let stored = store
168            .get_address("remote-only-bbbb".to_string())
169            .expect("remote address should have been stored");
170        assert_eq!(stored.street_address, "99 Remote Road");
171
172        // `apply` deliberately stamps last_sync with 0 - Desktop applies without
173        // telling us the server timestamp and sends it separately afterwards.
174        assert_eq!(bridge.last_sync().unwrap(), 0);
175        bridge.set_uploaded(1234, vec![local.guid.clone()]).unwrap();
176        bridge.sync_finished().unwrap();
177        assert_eq!(bridge.last_sync().unwrap(), 1234);
178
179        // Acknowledging the upload cleared the record's change counter, so a
180        // subsequent sync has nothing to send.
181        assert!(bridge.apply().expect("should apply again").is_empty());
182    }
